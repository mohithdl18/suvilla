export default function RoomAvailability({
    searchData = null,
    searchVersion = 0,
}) {
    const [availableRoomIds, setAvailableRoomIds] = useState([]);
    const [availability, setAvailability] = useState(null);

    const [loading, setLoading] = useState(false);
    const [error, setError] = useState("");

    const nights = searchData?.nights || 1;

    // ============================================================
    // CONVERT DATE TO YYYY-MM-DD
    // ============================================================

    const formatDateForAPI = (value) => {
        if (!value) return null;

        const date = value instanceof Date
            ? value
            : new Date(value);

        if (Number.isNaN(date.getTime())) {
            return null;
        }

        const year = date.getFullYear();
        const month = String(date.getMonth() + 1).padStart(2, "0");
        const day = String(date.getDate()).padStart(2, "0");

        return `${year}-${month}-${day}`;
    };

    // ============================================================
    // FETCH ROOM AVAILABILITY
    // ============================================================

    useEffect(() => {
        if (!searchData?.checkIn || !searchData?.checkOut) {
            setAvailableRoomIds([]);
            setAvailability(null);
            setError("");
            return;
        }

        const fetchAvailability = async () => {
            setLoading(true);
            setError("");

            try {
                const checkIn = formatDateForAPI(searchData.checkIn);
                const checkOut = formatDateForAPI(searchData.checkOut);

                if (!checkIn || !checkOut) {
                    throw new Error(
                        "Invalid check-in or check-out date."
                    );
                }

                const API_URL =
                    process.env.NEXT_PUBLIC_API_URL ||
                    "http://127.0.0.1:8000";

                const response = await fetch(
                    `${API_URL}/rooms/availability?check_in=${checkIn}&check_out=${checkOut}`,
                    {
                        method: "GET",
                        cache: "no-store",
                    }
                );

                const data = await response.json();

                if (!response.ok) {
                    throw new Error(
                        data?.detail ||
                        "Unable to check room availability."
                    );
                }

                const result = data?.availability;

                if (!result) {
                    throw new Error(
                        "Invalid availability response from server."
                    );
                }

                setAvailability(result);

                setAvailableRoomIds(
                    (result.available_rooms || []).map(
                        (room) => String(room.id)
                    )
                );
            } catch (err) {
                console.error(
                    "Room availability error:",
                    err
                );

                setAvailability(null);
                setAvailableRoomIds([]);

                setError(
                    err.message ||
                    "Unable to check room availability."
                );
            } finally {
                setLoading(false);
            }
        };

        fetchAvailability();
    }, [searchData, searchVersion]);

    // ============================================================
    // PRICE BREAKDOWN
    // ============================================================

    const getPriceBreakdown = (room) => {
        const roomPrice = room.price * nights;
        const taxes = Math.round(roomPrice * 0.12);
        const total = roomPrice + taxes;

        return {
            roomPrice,
            taxes,
            total,
        };
    };

    // ============================================================
    // INITIAL STATE
    // ============================================================

    if (!searchData) {
        return (
            <div className="w-full border border-foreground/10 bg-white px-6 py-12 text-center">
                <p className="font-body text-sm text-foreground/50">
                    Select your check-in and check-out dates
                    to see available rooms.
                </p>
            </div>
        );
    }

    // ============================================================
    // LOADING
    // ============================================================

    if (loading) {
        return (
            <div className="w-full border border-foreground/10 bg-white px-6 py-12 text-center">
                <p className="font-body text-sm text-foreground/50">
                    Checking room availability...
                </p>
            </div>
        );
    }

    // ============================================================
    // ERROR
    // ============================================================

    if (error) {
        return (
            <div className="w-full border border-red-200 bg-white px-6 py-12 text-center">
                <p className="font-body text-sm text-red-600">
                    {error}
                </p>

                <p className="mt-2 font-body text-xs text-foreground/40">
                    Please try searching again.
                </p>
            </div>
        );
    }

    // ============================================================
    // NO ROOMS AVAILABLE
    // ============================================================

    if (availableRoomIds.length === 0) {
        return (
            <div className="w-full border border-foreground/10 bg-white px-6 py-12 text-center">
                <p className="font-heading text-xl">
                    No rooms available
                </p>

                <p className="mt-2 font-body text-sm text-foreground/50">
                    All rooms are currently booked for
                    your selected dates.
                </p>
            </div>
        );
    }

    // ============================================================
    // AVAILABLE ROOMS
    // ============================================================

    return (
        <div className="w-full space-y-5">

            {rooms
                .filter((room) =>
                    availableRoomIds.includes(
                        String(room.id)
                    )
                )
                .map((room) => {

                    const price =
                        getPriceBreakdown(room);

                    return (
                        <article
                            key={room.id}
                            className="w-full overflow-hidden border border-foreground/15 bg-white"
                        >

                            <div className="grid grid-cols-1 lg:grid-cols-[280px_minmax(0,1fr)_260px]">

                                {/* =================================================
                                    LEFT — IMAGE + ROOM NAME ONLY
                                ================================================= */}

                                <div className="flex flex-col">

                                    {/* Image */}

                                    <div className="relative aspect-[16/10] w-full overflow-hidden bg-foreground/5 lg:aspect-auto lg:h-[190px]">

                                        <img
                                            src={room.image}
                                            alt={room.name}
                                            className="h-full w-full object-cover"
                                        />

                                    </div>

                                    {/* Room Name */}

                                    <div className="px-5 py-4">

                                        <p className="mb-1 font-body text-[9px] uppercase tracking-[0.2em] text-accent">
                                            Stay
                                        </p>

                                        <h3 className="font-heading text-xl leading-tight">
                                            {room.name}
                                        </h3>

                                    </div>

                                </div>


                                {/* =================================================
                                    MIDDLE — ROOM DETAILS + AMENITIES
                                ================================================= */}

                                <div className="border-t border-foreground/10 p-5 lg:border-l lg:border-t-0 lg:p-7">

                                    {/* Room details */}

                                    <div className="grid grid-cols-2 gap-x-6 gap-y-5 sm:grid-cols-4">

                                        <div>
                                            <p className="font-body text-[9px] uppercase tracking-[0.15em] text-foreground/40">
                                                Guests
                                            </p>

                                            <p className="mt-1 font-body text-sm">
                                                {room.guests}
                                            </p>
                                        </div>

                                        <div>
                                            <p className="font-body text-[9px] uppercase tracking-[0.15em] text-foreground/40">
                                                Bed
                                            </p>

                                            <p className="mt-1 font-body text-sm">
                                                {room.bed}
                                            </p>
                                        </div>

                                        <div>
                                            <p className="font-body text-[9px] uppercase tracking-[0.15em] text-foreground/40">
                                                Size
                                            </p>

                                            <p className="mt-1 font-body text-sm">
                                                {room.size}
                                            </p>
                                        </div>

                                        <div>
                                            <p className="font-body text-[9px] uppercase tracking-[0.15em] text-foreground/40">
                                                View
                                            </p>

                                            <p className="mt-1 font-body text-sm">
                                                {room.view}
                                            </p>
                                        </div>

                                    </div>


                                    {/* Divider */}

                                    <div className="my-6 border-t border-foreground/10" />


                                    {/* Amenities */}

                                    <div>

                                        <p className="font-body text-[10px] uppercase tracking-[0.2em] text-foreground/40">
                                            Amenities
                                        </p>

                                        <div className="mt-4 grid grid-cols-1 gap-y-3 sm:grid-cols-2">

                                            {room.amenities.map(
                                                (amenity) => (
                                                    <div
                                                        key={amenity}
                                                        className="flex items-center gap-3 font-body text-sm text-foreground/75"
                                                    >

                                                        <span className="flex h-5 w-5 shrink-0 items-center justify-center rounded-full border border-accent text-[10px] text-accent">
                                                            ✓
                                                        </span>

                                                        <span>
                                                            {amenity}
                                                        </span>

                                                    </div>
                                                )
                                            )}

                                        </div>

                                    </div>

                                </div>


                                {/* =================================================
                                    RIGHT — PRICE + BOOK NOW
                                ================================================= */}

                                <div className="flex flex-col justify-between border-t border-foreground/10 p-5 lg:border-l lg:border-t-0 lg:p-7">

                                    <div>

                                        <p className="font-body text-[10px] uppercase tracking-[0.2em] text-foreground/40">
                                            Price
                                        </p>

                                        {/* Price per night */}

                                        <div className="mt-3">

                                            <span className="font-heading text-2xl">
                                                ₹
                                                {room.price.toLocaleString(
                                                    "en-IN"
                                                )}
                                            </span>

                                            <span className="ml-1 font-body text-xs text-foreground/50">
                                                / night
                                            </span>

                                        </div>


                                        {/* Price breakdown */}

                                        <div className="mt-5 space-y-3 border-t border-foreground/10 pt-4">

                                            <div className="flex justify-between gap-4 font-body text-xs text-foreground/60">

                                                <span>
                                                    Room × {nights}{" "}
                                                    {nights === 1
                                                        ? "night"
                                                        : "nights"}
                                                </span>

                                                <span className="shrink-0">
                                                    ₹
                                                    {price.roomPrice.toLocaleString(
                                                        "en-IN"
                                                    )}
                                                </span>

                                            </div>


                                            <div className="flex justify-between gap-4 font-body text-xs text-foreground/60">

                                                <span>
                                                    Taxes & fees
                                                </span>

                                                <span className="shrink-0">
                                                    ₹
                                                    {price.taxes.toLocaleString(
                                                        "en-IN"
                                                    )}
                                                </span>

                                            </div>


                                            <div className="flex justify-between border-t border-foreground/10 pt-3 font-body text-sm font-medium">

                                                <span>
                                                    Total
                                                </span>

                                                <span>
                                                    ₹
                                                    {price.total.toLocaleString(
                                                        "en-IN"
                                                    )}
                                                </span>

                                            </div>

                                        </div>

                                    </div>


                                    {/* Book Now */}

                                    <button
                                        type="button"
                                        className="mt-7 flex w-full items-center justify-center gap-3 bg-foreground px-5 py-3.5 font-body text-xs font-medium uppercase tracking-wider text-primary transition hover:bg-accent hover:text-foreground"
                                    >
                                        Book Now

                                        <span className="text-base">
                                            →
                                        </span>

                                    </button>

                                </div>

                            </div>

                        </article>
                    );
                })}

        </div>
    );
}
