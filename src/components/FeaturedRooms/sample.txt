    const [activeIndex, setActiveIndex] = useState(0);

    const nextRoom = () => {
        setActiveIndex((prev) => (prev + 1) % rooms.length);
    };

    const previousRoom = () => {
        setActiveIndex(
            (prev) => (prev - 1 + rooms.length) % rooms.length
        );
    };

    /*
     * Position of each card relative to the active card:
     *
     * 0 = FRONT
     * 1 = RIGHT
     * 2 = BACK
     * 3 = LEFT
     */
    const getPosition = (index) => {
        return (index - activeIndex + rooms.length) % rooms.length;
    };

    /*
     * 3D carousel positioning
     */
    const getCardStyle = (position) => {
        switch (position) {
            // FRONT
            case 0:
                return {
                    transform:
                        "translate3d(0, 0, 180px) rotateY(0deg) scale(1)",
                    opacity: 1,
                    zIndex: 40,
                };

            // RIGHT
            case 1:
                return {
                    transform:
                        "translate3d(48%, 0, -20px) rotateY(-72deg) scale(0.9)",
                    opacity: 0.9,
                    zIndex: 30,
                };

            // BACK
            case 2:
                return {
                    transform:
                        "translate3d(0, 0, -180px) rotateY(180deg) scale(0.82)",
                    opacity: 0,
                    zIndex: 10,
                };

            case 3:
                return {
                    transform:
                        "translate3d(-48%, 0, -20px) rotateY(72deg) scale(0.9)",
                    opacity: 0.9,
                    zIndex: 30,
                };

            default:
                return {};
        }

                <p className="text-xs uppercase tracking-[0.3em] text-[#e4b441]">
                    Stay With Us

                    Our Rooms
                </h2>

            </div>


            {/* =====================================================
                CAROUSEL
            ====================================================== */}
            <div className="relative mt-16">

                {/* 3D Perspective */}
                <div
                    className="relative mx-auto h-[650px] w-full max-w-[1600px]"
                    style={{
                        perspective: "1400px",
                        perspectiveOrigin: "50% 50%",
                    }}
                >

                    {/* Carousel Stage */}
                    <div
                        className="absolute inset-0 flex items-center justify-center"
                        style={{
                            transformStyle: "preserve-3d",
                        }}
                    >

                        {rooms.map((room, index) => {

                            const position = getPosition(index);
                            const cardStyle = getCardStyle(position);

                            return (
                                <article
                                    key={room.id}
                                    className="absolute left-1/2 top-1/2 w-[680px] overflow-hidden rounded-t-[26px] bg-white shadow-2xl"

                                        ...cardStyle,

                                        /*
                                         * Fixed card dimensions
                                         */
                                        width: "680px",
                                        height: "600px",

                                        /*
                                         * Center card
                                         */
                                        marginLeft: "-340px",
                                        marginTop: "-300px",

                                        /*
                                         * 3D behaviour
                                         */
                                        transformStyle: "preserve-3d",
                                        backfaceVisibility: "hidden",

                                        /*
                                         * Smooth carousel movement
                                         */
                                        transition:
                                            "transform 900ms cubic-bezier(0.22, 1, 0.36, 1), opacity 700ms ease",
                                    }}
                                >

                                    {/* =================================================
                                        IMAGE
                                    ================================================== */}
                                    <div className="relative h-[280px] w-full overflow-hidden rounded-t-[26px] bg-[#052a50]">

                                        <img
                                            src={room.image}
                                            alt={room.name}
                                            className="h-full w-full object-cover transition-transform duration-700"
                                        />

                                        {/* Image Overlay */}

                                        CARD CONTENT
                                    ================================================== */}

                                        {/* Fixed Header */}

                                            <h3 className="text-2xl leading-tight text-[#052a50]">

                                            </h3>

                                        </div>


                                        {/* Fixed Description */}

                                            <p className="text-sm leading-7 text-[#052a50]/75">

                                            </p>

                                        </div>

                                                    ◉
                                                </span>

                                                {room.view}

                                            </div>

                                        </div>


                                        {/* =================================================
                                            AMENITIES
                                        ================================================== */}

                                            <div className="flex flex-wrap gap-2">

                                                {room.amenities.map(

                                                    )

                                        </div>

                                    </div>

                                </article>
                            );
                        })}

                </div>

                <div className="relative z-50 flex justify-center gap-3">
